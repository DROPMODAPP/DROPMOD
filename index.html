import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, }); Deno.serve(async (req) => { // ============================================================ // CORS // ============================================================ if (req.method === "OPTIONS") { return new Response("ok", { status: 200, headers: corsHeaders, }); } try { if (req.method !== "POST") { return json( { error: "Método não permitido" }, 405 ); } const body = await req.json(); const action = String( body.action ?? "login" ) .trim() .toLowerCase(); const username = String( body.username ?? "" ) .trim() .toLowerCase(); const password = String( body.password ?? "" ); const name = String( body.name ?? body.username ?? "" ).trim(); if (!username) { return json( { error: "Usuário é obrigatório", }, 400 ); } if ( (action === "login" || action === "create_operator" || action === "reset_operator_password") && password.length < 6 ) { return json( { error: "A senha deve ter pelo menos 6 caracteres", }, 400 ); } // ============================================================ // CONFIGURAÇÃO DO SUPABASE // ============================================================ const supabaseUrl = Deno.env.get("SUPABASE_URL"); const serviceRoleKey = Deno.env.get( "SUPABASE_SERVICE_ROLE_KEY" ); const anonKey = Deno.env.get( "SUPABASE_ANON_KEY" ); if ( !supabaseUrl || !serviceRoleKey || !anonKey ) { return json( { error: "Configuração interna do Supabase incompleta.", }, 500 ); } const supabaseAdmin = createClient( supabaseUrl, serviceRoleKey ); // ============================================================ // LOGIN // ============================================================ if (action === "login") { /* * O DROPMOD utiliza a tabela: * * public.perfis * * Campos: * id * empresa_id * nome * usuario * tipo * ativo */ const { data: perfil, error: perfilError, } = await supabaseAdmin .from("perfis") .select( "id, empresa_id, nome, usuario, tipo, ativo" ) .eq("usuario", username) .maybeSingle(); if (perfilError) { console.error( "Erro ao consultar perfil:", perfilError ); return json( { error: "Erro ao consultar cadastro do usuário", }, 500 ); } if (!perfil) { return json( { error: "Usuário ou senha inválidos", }, 401 ); } // Conta desativada if (perfil.ativo === false) { return json( { error: "Este usuário está desativado", }, 403 ); } // ========================================================== // LOCALIZA CONTA NO SUPABASE AUTH // ========================================================== const { data: authUserData, error: authUserError, } = await supabaseAdmin.auth.admin.getUserById( perfil.id ); const email = authUserData?.user?.email; if ( authUserError || !email ) { return json( { error: "Conta de autenticação não encontrada", }, 401 ); } // ========================================================== // AUTENTICAÇÃO REAL // ========================================================== const authClient = createClient( supabaseUrl, anonKey ); const { data: loginData, error: loginError, } = await authClient.auth.signInWithPassword( { email, password, } ); if ( loginError || !loginData.session ) { console.error( "Erro de autenticação:", loginError ); return json( { error: "Usuário ou senha inválidos", }, 401 ); } // ========================================================== // RETORNO PARA O DROPMOD PÚBLICO // ========================================================== return json({ success: true, session: { access_token: loginData.session .access_token, refresh_token: loginData.session .refresh_token, }, profile: { id: perfil.id, empresa_id: perfil.empresa_id, organization_id: perfil.empresa_id, nome: perfil.nome, name: perfil.nome, usuario: perfil.usuario, username: perfil.usuario, tipo: perfil.tipo, // Retorna o papel real. O App Público decide o fluxo // com base em "tipo", e não transforma operador em admin. role: String(perfil.tipo || "").trim().toUpperCase() === "OPERADOR" ? "operador" : "admin", ativo: perfil.ativo, }, }); } // ============================================================ // AÇÕES ADMINISTRATIVAS SOBRE OPERADORES // ============================================================ if ( action === "update_operator_name" || action === "reset_operator_password" ) { const authHeader = req.headers.get("Authorization") ?? ""; if (!authHeader) { return json( { error: "Usuário não autenticado" }, 401 ); } const supabaseUser = createClient( supabaseUrl, anonKey, { global: { headers: { Authorization: authHeader, }, }, } ); const { data: { user }, error: userError, } = await supabaseUser.auth.getUser(); if (userError || !user) { return json( { error: "Usuário não autenticado" }, 401 ); } const { data: adminPerfil, error: adminPerfilError, } = await supabaseAdmin .from("perfis") .select( "id, empresa_id, nome, usuario, tipo, ativo" ) .eq("id", user.id) .maybeSingle(); if ( adminPerfilError || !adminPerfil ) { return json( { error: "Perfil do administrador não encontrado" }, 403 ); } const adminTipo = String( adminPerfil.tipo ?? "" ) .trim() .toUpperCase(); if ( !( adminTipo === "ADM" || adminTipo === "ADMIN" || adminTipo === "ADMINISTRADOR" ) ) { return json( { error: "Apenas administradores podem gerenciar operadores" }, 403 ); } if (!adminPerfil.empresa_id) { return json( { error: "Administrador não está vinculado a uma empresa" }, 403 ); } const targetUsername = String(body.username ?? "") .trim() .toLowerCase(); if (!targetUsername) { return json( { error: "Usuário do operador é obrigatório" }, 400 ); } const { data: operador, error: operadorError, } = await supabaseAdmin .from("perfis") .select( "id, empresa_id, nome, usuario, tipo, ativo" ) .eq("usuario", targetUsername) .eq("empresa_id", adminPerfil.empresa_id) .maybeSingle(); if (operadorError || !operador) { return json( { error: "Operador não encontrado nesta empresa" }, 404 ); } if ( String(operador.tipo ?? "").trim().toUpperCase() !== "OPERADOR" ) { return json( { error: "O perfil informado não é um operador" }, 403 ); } if (action === "update_operator_name") { const novoNome = String(body.name ?? "") .trim(); if (!novoNome) { return json( { error: "O novo nome é obrigatório" }, 400 ); } const { error: updateError } = await supabaseAdmin .from("perfis") .update({ nome: novoNome, }) .eq("id", operador.id) .eq("empresa_id", adminPerfil.empresa_id); if (updateError) { console.error( "Erro ao alterar nome do operador:", updateError ); return json( { error: updateError.message }, 400 ); } return json({ success: true, message: "Nome do operador alterado com sucesso", }); } const { error: resetError } = await supabaseAdmin.auth.admin.updateUserById( operador.id, { password, } ); if (resetError) { console.error( "Erro ao alterar senha do operador:", resetError ); return json( { error: resetError.message }, 400 ); } return json({ success: true, message: "Senha do operador alterada com sucesso", }); } // ============================================================ // CRIAÇÃO DE OPERADOR // ============================================================ if ( action !== "create_operator" ) { return json( { error: "Ação inválida", }, 400 ); } // ============================================================ // VERIFICA AUTORIZAÇÃO DO ADM // ============================================================ const authHeader = req.headers.get( "Authorization" ) ?? ""; if (!authHeader) { return json( { error: "Usuário não autenticado", }, 401 ); } const supabaseUser = createClient( supabaseUrl, anonKey, { global: { headers: { Authorization: authHeader, }, }, } ); const { data: { user, }, error: userError, } = await supabaseUser.auth.getUser(); if ( userError || !user ) { return json( { error: "Usuário não autenticado", }, 401 ); } // ============================================================ // LOCALIZA ADM NA TABELA CORRETA // ============================================================ const { data: adminPerfil, error: adminPerfilError, } = await supabaseAdmin .from("perfis") .select( "id, empresa_id, nome, usuario, tipo, ativo" ) .eq("id", user.id) .maybeSingle(); if ( adminPerfilError || !adminPerfil ) { return json( { error: "Perfil do administrador não encontrado", }, 403 ); } const adminTipo = String( adminPerfil.tipo ?? "" ) .trim() .toUpperCase(); const adminIsAdmin = adminTipo === "ADM" || adminTipo === "ADMIN" || adminTipo === "ADMINISTRADOR"; if (!adminIsAdmin) { return json( { error: "Apenas administradores podem criar operadores", }, 403 ); } if ( !adminPerfil.empresa_id ) { return json( { error: "Administrador não está vinculado a uma empresa", }, 403 ); } // ============================================================ // VALIDAÇÃO DO USUÁRIO DO OPERADOR // ============================================================ if ( !/^[a-z0-9._-]{3,40}$/.test( username ) ) { return json( { error: "Usuário inválido. Use 3 a 40 caracteres: letras, números, ponto, hífen ou underline, sem espaços.", }, 400 ); } // ============================================================ // E-MAIL INTERNO DO OPERADOR // ============================================================ const email = `u.${username}.${String( adminPerfil.empresa_id ).replaceAll("-", "")}@dropmod.local`; // ============================================================ // CRIA USUÁRIO NO SUPABASE AUTH // ============================================================ const { data: createdUser, error: createError, } = await supabaseAdmin.auth.admin.createUser( { email, password, email_confirm: true, user_metadata: { name, username, empresa_id: adminPerfil.empresa_id, tipo: "OPERADOR", }, } ); if (createError) { console.error( "Erro ao criar operador:", createError ); return json( { error: createError.message, }, 400 ); } // ============================================================ // CRIA PERFIL DO OPERADOR // ============================================================ const { error: perfilInsertError, } = await supabaseAdmin .from("perfis") .insert({ id: createdUser.user.id, empresa_id: adminPerfil.empresa_id, nome: name || username, usuario: username, tipo: "OPERADOR", ativo: true, }); if (perfilInsertError) { console.error( "Erro ao criar perfil:", perfilInsertError ); // Desfaz usuário Auth await supabaseAdmin.auth.admin.deleteUser( createdUser.user.id ); return json( { error: "Não foi possível criar o perfil do operador: " + perfilInsertError.message, }, 500 ); } // ============================================================ // SUCESSO // ============================================================ return json({ success: true, message: "Operador criado com sucesso", user_id: createdUser.user.id, username, empresa_id: adminPerfil.empresa_id, }); } catch (error) { console.error( "quick-task error:", error ); return json( { error: error instanceof Error ? error.message : "Erro interno", }, 500 ); } });
